### Código do teste
```js
const { calculateDiscount } = require('../pricing');

describe('calculateDiscount', () => {
  it('aplica desconto percentual corretamente sobre um preço válido', () => {
    expect(calculateDiscount(100, 10)).toBe(90);
  });

  it('retorna o preço original quando o desconto é 0%', () => {
    expect(calculateDiscount(100, 0)).toBe(100);
  });

  it('lida com preço igual a zero', () => {
    expect(calculateDiscount(0, 10)).toBe(0);
  });

  it('lida com desconto de 100%, retornando zero', () => {
    expect(calculateDiscount(100, 100)).toBe(0);
  });

  it('lida com preço com casas decimais longas', () => {
    expect(calculateDiscount(19.99, 10)).toBeCloseTo(17.991, 3);
  });

  it('lança erro quando o preço é negativo', () => {
    expect(() => calculateDiscount(-10, 10)).toThrow('price must be non-negative');
  });

  it('lança erro quando o desconto é negativo', () => {
    expect(() => calculateDiscount(100, -5)).toThrow('discountPercent must be between 0 and 100');
  });

  it('lança erro quando o desconto é maior que 100', () => {
    expect(() => calculateDiscount(100, 150)).toThrow('discountPercent must be between 0 and 100');
  });
});
```
